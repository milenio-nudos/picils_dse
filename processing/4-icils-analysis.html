3.1.3 Table processing

# Dataframe consolidado

tabla_completa_cnt <- data.frame(
  Type = c("Country", "", ""),
  Model = c("1. Configural", "2. Metric", "3. Scalar"),
  chisq = c(fitmeasures_config_cnt["chisq"], fitmeasures_metric_cnt["chisq"], fitmeasures_scalar_cnt["chisq"]),
  df = c(fitmeasures_config_cnt["df"], fitmeasures_metric_cnt["df"], fitmeasures_scalar_cnt["df"]),
  CFI = c(fitmeasures_config_cnt["cfi"], fitmeasures_metric_cnt["cfi"], fitmeasures_scalar_cnt["cfi"]),
  TLI = c(fitmeasures_config_cnt["tli"], fitmeasures_metric_cnt["tli"], fitmeasures_scalar_cnt["tli"]),
  RMSEA = c(fitmeasures_config_cnt["rmsea"], fitmeasures_metric_cnt["rmsea"], fitmeasures_scalar_cnt["rmsea"])
)

# Cálculo deltas

tabla_completa_cnt$delta_chisq <- c(NA, diff(tabla_completa_cnt$chisq))
tabla_completa_cnt$delta_df    <- c(NA, diff(tabla_completa_cnt$df))
tabla_completa_cnt$delta_CFI   <- c(NA, diff(tabla_completa_cnt$CFI))
tabla_completa_cnt$delta_RMSEA <- c(NA, diff(tabla_completa_cnt$RMSEA))

# Display para los deltas 

tabla_completa_cnt$delta_CFI_display <- sapply(1:nrow(tabla_completa_cnt), function(i) {
  if (i == 1) return(NA)
  
  if (tabla_completa_cnt$delta_CFI[i] >= -0.004) {
    return("< -0.004")
  } else {
    return(sprintf("%.3f", tabla_completa_cnt$delta_CFI[i]))
  }
})

tabla_completa_cnt$delta_RMSEA_display <- sapply(1:nrow(tabla_completa_cnt), function(i) {
  if (i == 1) return(NA)
  
  umbral <- if (i == 2) 0.050 else 0.010
  
  if (tabla_completa_cnt$delta_RMSEA[i] <= umbral) {
    return(paste0("< ", umbral))
  } else {
    return(sprintf("%.3f", tabla_completa_cnt$delta_RMSEA[i]))
  }
})

# ANOVA para significancia

anova_cnt <- anova(config_cnt, metric_cnt, scalar_cnt)

tabla_completa_cnt$p_value_delta_chisq <- case_when(anova_cnt$`Pr(>Chisq)`< 0.01 ~ "< 0.01",
                                                  anova_cnt$`Pr(>Chisq)`< 0.05 ~ "< 0.05",
                                                  anova_cnt$`Pr(>Chisq)`> 0.05 ~ sprintf("%.2f", anova_cnt$`Pr(>Chisq)`))
tabla_completa_cnt$Decision <- c(
  NA,  # Configural no tiene comparación previa
  # Metric: ΔCFI ≥ -0.004 Y ΔRMSEA ≤ 0.050
  ifelse(tabla_completa_cnt$delta_CFI[2] >= -0.004 & 
         tabla_completa_cnt$delta_RMSEA[2] <= 0.050, "Yes", "No"),
  # Scalar: ΔCFI ≥ -0.004 Y ΔRMSEA ≤ 0.010
  ifelse(tabla_completa_cnt$delta_CFI[3] >= -0.004 & 
         tabla_completa_cnt$delta_RMSEA[3] <= 0.010, "Yes", "No")
)

3.3 Final table formatting

tabla_completa_cnt_export <- tabla_completa_cnt %>%
  select(Type, Model, chisq, df, CFI, TLI, RMSEA,
         delta_chisq, delta_df, 
         delta_CFI_display, delta_RMSEA_display,
         p_value_delta_chisq, Decision) %>%
  rename(delta_CFI = delta_CFI_display,
         delta_RMSEA = delta_RMSEA_display)

tabla_completa_sex_export <- tabla_completa_sex %>%
  select(Type, Model, chisq, df, CFI, TLI, RMSEA,
         delta_chisq, delta_df, 
         delta_CFI_display, delta_RMSEA_display,
         p_value_delta_chisq, Decision) %>%
  rename(delta_CFI = delta_CFI_display,
         delta_RMSEA = delta_RMSEA_display)

tabla_unificada_icils <- bind_rows(tabla_completa_cnt_export, tabla_completa_sex_export)
colnames(tabla_unificada_icils) <- c(
  "Type",
  "Model", 
  "\u03C7\u00B2",      # χ²
  "df",
  "CFI",
  "TLI",
  "RMSEA",
  "\u0394\u03C7\u00B2", # Δχ²
  "\u0394df",          # Δdf
  "\u0394CFI",         # ΔCFI
  "\u0394RMSEA",       # ΔRMSEA
  "p",
  "Decision"
)

ft_icils <- flextable(tabla_unificada_icils) %>%
# Formatear números: 0 decimales para χ², df, Δχ², Δdf
  colformat_double(j = c("\u03C7\u00B2", "df", "\u0394\u03C7\u00B2", "\u0394df"), digits = 0) %>%
  # 3 decimales para el resto de columnas numéricas
  colformat_double(j = c("CFI", "TLI", "RMSEA", "\u0394CFI", "\u0394RMSEA", "p"), digits = 3) %>%  border_remove() %>%
  hline_top(border = officer::fp_border(width = 2), part = "all") %>%
  hline_bottom(border = officer::fp_border(width = 2), part = "all") %>%
  hline(i = 1, border = officer::fp_border(width = 1.5), part = "header") %>%
  bold(part = "header") %>%
  font(fontname = "Calibri", part = "all") %>%
  fontsize(size = 9, part = "body") %>%
  fontsize(size = 10, part = "header") %>%
  padding(padding = 3, part = "all") %>%
  align(align = "center", part = "header") %>%
  align(align = "left", part = "body", j = 1:2) %>%
  align(align = "center", part = "body", j = 3:ncol(tabla_unificada_icils)) %>%
  autofit() %>%
  fit_to_width(max_width = 7.5)

# --- LOGICA DE COLOREO AUTOMÁTICA --- #

# Detectamos la columna a colorear
delta_cfi_col <- which(colnames(tabla_unificada_icils) == "\u0394CFI")
delta_rmsea_col <- which(colnames(tabla_unificada_icils) == "\u0394RMSEA")
decision_col <- which(colnames(tabla_unificada_icils) == "Decision")
# Detectamos la fila a colorear (las que no tienen el operador menor que)
rows_cfi_fail <- which(!grepl("<", tabla_unificada_icils[[delta_cfi_col]]) & 
                       !is.na(tabla_unificada_icils[[delta_cfi_col]]))
rows_rmsea_fail <- which(!grepl("<", tabla_unificada_icils[[delta_rmsea_col]]) & 
                         !is.na(tabla_unificada_icils[[delta_rmsea_col]]))
rows_decision_no <- which(tabla_unificada_icils$Decision == "No")
# Coloreamos según corresponda
if(length(rows_cfi_fail) > 0) {
  ft_icils <- ft_icils %>%
    bg(i = rows_cfi_fail, j = delta_cfi_col, bg = "#FFCCCB") %>%
    color(i = rows_cfi_fail, j = delta_cfi_col, color = "#8B0000")
}

if(length(rows_rmsea_fail) > 0) {
  ft_icils <- ft_icils %>%
    bg(i = rows_rmsea_fail, j = delta_rmsea_col, bg = "#FFCCCB") %>%
    color(i = rows_rmsea_fail, j = delta_rmsea_col, color = "#8B0000")
}

if(length(rows_decision_no) > 0) {
  ft_icils <- ft_icils %>%
    bg(i = rows_decision_no, j = decision_col, bg = "#FFCCCB") %>%
    color(i = rows_decision_no, j = decision_col, color = "#8B0000") %>%
    bold(i = rows_decision_no, j = decision_col)
}
