2.1 Model running and data processing

Code 
pisa <- pisa |>
  mutate(
    effgen = rowMeans(across(c(search_info, evaluate_info, 
                               explain_content, share_content, pair_collab,
                               write_text)), 
                      na.rm = TRUE),
    effspec = rowMeans(across(c(programming, identify_error, logical_solution)), 
                       na.rm = TRUE)
  )

model_fitted_p <- '
  gen_dse =~ search_info + evaluate_info + 
  explain_content + share_content + pair_collab + 
  write_text
  spec_dse =~ programming + identify_error + logical_solution
  '
  
fitgroup_p <- cfa(model = model_fitted_p, 
                data = pisa, 
                group = "CNT",
                ordered = TRUE,
                estimator = "DWLS"
              )

2.5 Gender DSE differences by country

Code 
gender_plot <- pisa |>  
  group_by(CNT, sex) |>  
  summarise(
    "General DSE" = mean(effgen, na.rm = TRUE),
    "Specialized DSE" = mean(effspec, na.rm = TRUE)
  ) |>  
  pivot_longer(
    cols = 3:4,
    names_to = "type",
    values_to = "score"
  ) |>  
  ungroup() |>  
  mutate(sex = to_label(sex)) |>  
  drop_na()

argument.

4.1 Function for HD saving

Code 
export_plot <- function(plot_object, 
                       filename,
                       format = c("png", "pdf", "tiff", "eps", "svg"),
                       width = 10,
                       height = 7,
                       dpi = 300,
                       units = "in",
                       bg = "white",
                       path = "./output/figures/") {
  
  format <- match.arg(format)
  
  full_filename <- paste0(filename, ".", format)
  full_path <- file.path(path, full_filename)
  
  # Exportar según formato
  ggsave(
    filename = full_path,
    plot = plot_object,
    width = width,
    height = height,
    dpi = dpi,
    units = units,
    bg = bg,
    device = format
  )
  
  message(paste0("✓ Gráfico exportado: ", full_path))
}

4.2 Function for multiformat HD saving

Code 
export_plot_multi <- function(plot_object,
                              filename,
                              formats = c("png", "pdf", "tiff"),
                              width = 10,
                              height = 7,
                              dpi = 300,
                              units = "in",
                              bg = "white",
                              path = "./output/figures/") {
  
  purrr::walk(formats, ~export_plot(
    plot_object = plot_object,
    filename = filename,
    format = .x,
    width = width,
    height = height,
    dpi = dpi,
    units = units,
    bg = bg,
    path = path
  ))
}
