2 Pooled model

model_syntax <- '
  gen_dse =~ search_info + evaluate_info + collect_data + 
  explain_content + share_content + pair_collab + 
  write_text + create_media + change_settings
  spec_dse =~ programming + develop_webpage + identify_error + identify_app + logical_solution

m_pooled <- cfa(model = model_syntax,

First, we detect a couple of problematic items. Identify app charge to the general factor with a mi of 76,502 (very high), and shows strong residual covariance with a lot of items (change settings, identify error, logical solution, programming). The item of develop webpage self-efficacy shows a big cross-loading to the general factor (mi=39,014). Looking at the residuals, it has strong residual covariance with collect data, create media and change settings, and a negative one with items from the specialized factor (i.e. logical solution and identify error). This item will be also be removed. Summing up to the technical reasons, the growing democratization of web development impulsed by the user-friendly platforms might have changed the perception of this item by the students in this cycle. In comparison with ICILS, this item refers to a webpage or a blog, it could be that students relate blogs with more non-technical platforms like fandom.

4.3.1 Residuals table

get_upper_tri <- function(matriz, model_name) {
  matriz[lower.tri(matriz, diag = TRUE)] <- NA
  as.data.frame(as.table(matriz)) %>%
    na.omit() %>%
    rename(Item1 = Var1, Item2 = Var2, Residual = Freq) %>%
    mutate(Model = model_name)
}

df_res_1 <- get_upper_tri(matriz_residuales_1, "Model 1 (Initial)")
df_res_2 <- get_upper_tri(matriz_residuales_2, "Model 2 (Re-specified)")

df_plot <- bind_rows(df_res_1, df_res_2)

cutoff <- 0.1 

table_residuals <- df_plot %>% 
  filter(abs(Residual) > cutoff) %>%
  arrange(Model, desc(abs(Residual))) %>%
  unite("Pair", Item1, Item2, sep = " \u223C\u223C ")

ft_resid <- flextable(table_residuals) %>%
  set_header_labels(
    Model = "Model Stage",
    Pair = "Item Pair",
    Residual = "Residual Corr."
  ) %>%

  merge_v(j = "Model") %>%
  colformat_double(j = "Residual", digits = 3) %>%
  font(fontname = "Times New Roman", part = "all") %>%
  fontsize(size = 10, part = "all") %>%
  bold(part = "header") %>%

  border_remove() %>%
  hline_top(border = fp_border(width = 2), part = "all") %>%
  hline_bottom(border = fp_border(width = 2), part = "body") %>%
  hline(part = "header", border = fp_border(width = 1.5)) %>%
  hline(i = ~ !duplicated(Model, fromLast = TRUE), border = fp_border(color = "grey", width = 0.5)) %>%
  
  autofit() %>%
  add_footer_lines(values = paste("Note. Only pairs with residual correlations >", cutoff, "are shown.")) %>%
  fontsize(size = 9, part = "footer") %>% 
  italic(part = "footer")

# Deltas display

tabla_completa_cnt$delta_CFI_display <- sapply(1:nrow(tabla_completa_cnt), function(i) {
  if (i == 1) return(NA)
  
  if (tabla_completa_cnt$delta_CFI[i] >= -0.004) {
    return("< -0.004")
  } else {
    return(sprintf("%.3f", tabla_completa_cnt$delta_CFI[i]))
  }
})

tabla_completa_cnt$delta_RMSEA_display <- sapply(1:nrow(tabla_completa_cnt), function(i) {
  if (i == 1) return(NA)
  
  umbral <- if (i == 2) 0.050 else 0.010
  
  if (tabla_completa_cnt$delta_RMSEA[i] <= umbral) {
    return(paste0("< ", umbral))
  } else {
    return(sprintf("%.3f", tabla_completa_cnt$delta_RMSEA[i]))
  }
})

6.4 Function for HD saving

export_plot <- function(plot_object, 
                       filename,
                       format = c("png", "pdf", "tiff", "eps", "svg"),
                       width = 10,
                       height = 7,
                       dpi = 300,
                       units = "in",
                       bg = "white",
                       path = "./output/figures/") {
  
  format <- match.arg(format)
  
  full_filename <- paste0(filename, ".", format)
  full_path <- file.path(path, full_filename)
  

  ggsave(
    filename = full_path,
    plot = plot_object,
    width = width,
    height = height,
    dpi = dpi,
    units = units,
    bg = bg,
    device = format
  )
  
  message(paste0("✓ Gráfico exportado: ", full_path))
}
