model_syntax <- '
  gen_dse =~ search_info + evaluate_info + collect_data +

First, we detect a couple of problematic items. Identify app charge to the general factor with a mi of 76,502 (very high), and shows strong residual covariance with a lot of items (change settings, identify error, logical solution, programming). We decide to remove that item, because is causing too much noise in the model and possibly dont adjust with the programing-related tasks of the specialized DSE factor.

The item of develop webpage self-efficacy shows a big cross-loading to the general factor (mi=39,014). Looking at the residuals, it has strong residual covariance with collect data, create media and change settings, and a negative one with items from the specialized factor (i.e. logical solution and identify error). This item will be also be removed. Summing up to the technical reasons, the growing democratization of web development impulsed by the user-friendly platforms might have changed the perception of this item by the students in this cycle. In comparison with ICILS, this item refers to a webpage or a blog, it could be that students relate blogs with more non-technical platforms like fandom.

5.1.2 Table processing

tabla_completa_cnt <- data.frame(
  Type = c("Country", "", ""),
  Model = c("1. Configural", "2. Metric", "3. Scalar"),
  chisq = c(fitmeasures_config_cnt["chisq"], fitmeasures_metric_cnt["chisq"], fitmeasures_scalar_cnt["chisq"]),
  df = c(fitmeasures_config_cnt["df"], fitmeasures_metric_cnt["df"], fitmeasures_scalar_cnt["df"]),
  CFI = c(fitmeasures_config_cnt["cfi"], fitmeasures_metric_cnt["cfi"], fitmeasures_scalar_cnt["cfi"]),
  TLI = c(fitmeasures_config_cnt["tli"], fitmeasures_metric_cnt["tli"], fitmeasures_scalar_cnt["tli"]),
  RMSEA = c(fitmeasures_config_cnt["rmsea"], fitmeasures_metric_cnt["rmsea"], fitmeasures_scalar_cnt["rmsea"])
)

# Deltas

tabla_completa_cnt$delta_chisq <- c(NA, diff(tabla_completa_cnt$chisq))
tabla_completa_cnt$delta_df    <- c(NA, diff(tabla_completa_cnt$df))
tabla_completa_cnt$delta_CFI   <- c(NA, diff(tabla_completa_cnt$CFI))
tabla_completa_cnt$delta_RMSEA <- c(NA, diff(tabla_completa_cnt$RMSEA))

# Deltas display

tabla_completa_cnt$delta_CFI_display <- sapply(1:nrow(tabla_completa_cnt), function(i) {
  if (i == 1) return(NA)
  
  if (tabla_completa_cnt$delta_CFI[i] >= -0.004) {
    return("< -0.004")
  } else {
    return(sprintf("%.3f", tabla_completa_cnt$delta_CFI[i]))
  }
})

tabla_completa_cnt$delta_RMSEA_display <- sapply(1:nrow(tabla_completa_cnt), function(i) {
  if (i == 1) return(NA)
  
  umbral <- if (i == 2) 0.050 else 0.010
  
  if (tabla_completa_cnt$delta_RMSEA[i] <= umbral) {
    return(paste0("< ", umbral))
  } else {
    return(sprintf("%.3f", tabla_completa_cnt$delta_RMSEA[i]))
  }
})

# ANOVA for significance

anova_cnt <- anova(config_cnt, metric_cnt, scalar_cnt)

tabla_completa_cnt$p_value_delta_chisq <- case_when(anova_cnt$`Pr(>Chisq)`< 0.01 ~ "< 0.01",
                                                  anova_cnt$`Pr(>Chisq)`< 0.05 ~ "< 0.05",
                                                  anova_cnt$`Pr(>Chisq)`> 0.05 ~ sprintf("%.2f", anova_cnt$`Pr(>Chisq)`))
tabla_completa_cnt$Decision <- c(
  NA,  # Configural
  # Metric: ΔCFI ≥ -0.004 Y ΔRMSEA ≤ 0.050
  ifelse(tabla_completa_cnt$delta_CFI[2] >= -0.004 & 
         tabla_completa_cnt$delta_RMSEA[2] <= 0.050, "Yes", "No"),
  # Scalar: ΔCFI ≥ -0.004 Y ΔRMSEA ≤ 0.010
  ifelse(tabla_completa_cnt$delta_CFI[3] >= -0.004 & 
         tabla_completa_cnt$delta_RMSEA[3] <= 0.010, "Yes", "No")
)

6.4 Function for HD saving

export_plot <- function(plot_object, 
                       filename,
                       format = c("png", "pdf", "tiff", "eps", "svg"),
                       width = 10,
                       height = 7,
                       dpi = 300,
                       units = "in",
                       bg = "white",
                       path = "./output/figures/") {
  
  format <- match.arg(format)
  
  full_filename <- paste0(filename, ".", format)
  full_path <- file.path(path, full_filename)
  

  ggsave(
    filename = full_path,
    plot = plot_object,
    width = width,
    height = height,
    dpi = dpi,
    units = units,
    bg = bg,
    device = format
  )
  
  message(paste0("✓ Gráfico exportado: ", full_path))
}

6.5 Function for multiformat HD saving

export_plot_multi <- function(plot_object,
                              filename,
                              formats = c("png", "pdf", "tiff"),
                              width = 10,
                              height = 7,
                              dpi = 300,
                              units = "in",
                              bg = "white",
                              path = "./output/figures/") {
  
  purrr::walk(formats, ~export_plot(
    plot_object = plot_object,
    filename = filename,
    format = .x,
    width = width,
    height = height,
    dpi = dpi,
    units = units,
    bg = bg,
    path = path
  ))
}
