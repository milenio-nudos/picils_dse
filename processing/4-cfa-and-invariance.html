# Extracción de indicadores de ajuste

fitmeasures_config_p <- fitMeasures(multigroup_p,
                         c("chisq", "df", "pvalue", "cfi", "tli", "rmsea", "srmr"))

fitmeasures_metric_p <- fitMeasures(multigroup_p_m,
                         c("chisq", "df", "pvalue", "cfi", "tli", "rmsea", "srmr"))

fitmeasures_scalar_p <- fitMeasures(multigroup_p_e,
                         c("chisq", "df", "pvalue", "cfi", "tli", "rmsea", "srmr"))


# Tabla resumen (html y opción latex)

tabla_completa_p <- data.frame(
  Estudio = c("PISA", "PISA", "PISA"),
  Modelo = c("1. Configural", "2. Metric", "3. Scalar"),
  chisq = c(fitmeasures_config_p["chisq"], fitmeasures_metric_p["chisq"], fitmeasures_scalar_p["chisq"]),
  df = c(fitmeasures_config_p["df"], fitmeasures_metric_p["df"], fitmeasures_scalar_p["df"]),
  CFI = c(fitmeasures_config_p["cfi"], fitmeasures_metric_p["cfi"], fitmeasures_scalar_p["cfi"]),
  TLI = c(fitmeasures_config_p["tli"], fitmeasures_metric_p["tli"], fitmeasures_scalar_p["tli"]),
  RMSEA = c(fitmeasures_config_p["rmsea"], fitmeasures_metric_p["rmsea"], fitmeasures_scalar_p["rmsea"]),
  SRMR = c(fitmeasures_config_p["srmr"], fitmeasures_metric_p["srmr"], fitmeasures_scalar_p["srmr"])
)

# Cálculo deltas

tabla_completa_p$delta_chisq <- c(NA, diff(tabla_completa_p$chisq))
tabla_completa_p$delta_df    <- c(NA, diff(tabla_completa_p$df))
tabla_completa_p$delta_CFI   <- c(NA, diff(tabla_completa_p$CFI))
tabla_completa_p$delta_RMSEA <- c(NA, diff(tabla_completa_p$RMSEA))

# Display para los deltas 

tabla_completa_p$delta_CFI_display <- sapply(1:nrow(tabla_completa_p), function(i) {
  if (i == 1) return(NA)
  
  if (tabla_completa_p$delta_CFI[i] >= -0.004) {
    return("< -0.004")
  } else {
    return(sprintf("%.3f", tabla_completa_p$delta_CFI[i]))
  }
})

tabla_completa_p$delta_RMSEA_display <- sapply(1:nrow(tabla_completa_p), function(i) {
  if (i == 1) return(NA)
  
  umbral <- if (i == 2) 0.050 else 0.010
  
  if (tabla_completa_p$delta_RMSEA[i] <= umbral) {
    return(paste0("< ", umbral))
  } else {
    return(sprintf("%.3f", tabla_completa_p$delta_RMSEA[i]))
  }
})

# ANOVA para significancia

test_anova <- anova(multigroup_p, multigroup_p_m, multigroup_p_e)

tabla_completa_p$p_value_delta_chisq <- case_when(test_anova$`Pr(>Chisq)`< 0.01 ~ "< 0.01",
                                                  test_anova$`Pr(>Chisq)`< 0.05 ~ "< 0.05",
                                                  test_anova$`Pr(>Chisq)`> 0.05 ~ sprintf("%.2f", test_anova$`Pr(>Chisq)`))

tabla_completa_p$Decision <- c(
  NA,  # Configural no tiene comparación previa
  # Métrico: ΔCFI ≥ -0.004 Y ΔRMSEA ≤ 0.050
  ifelse(tabla_completa_p$delta_CFI[2] >= -0.004 & 
         tabla_completa_p$delta_RMSEA[2] <= 0.050, "Yes", "No"),
  # Escalar: ΔCFI ≥ -0.004 Y ΔRMSEA ≤ 0.010
  ifelse(tabla_completa_p$delta_CFI[3] >= -0.004 & 
         tabla_completa_p$delta_RMSEA[3] <= 0.010, "Yes", "No")
)
